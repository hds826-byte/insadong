1 Google Sheets 시트 구성

시트명: members 컬럼: id | name | dept | email | password | phone | intro | role | joinedAt

2 Apps Script 코드 추가

// doPost 내 'register' 액션 case 'register': { const sheet = ss.getSheetByName('members'); const email = body.email; // 중복 확인 const emails = sheet.getRange(2,4,sheet.getLastRow()).getValues().flat(); if (emails.includes(email)) return out({error:'이미 가입된 이메일입니다.'}); const id = Utilities.getUuid(); const hash = Utilities.computeDigest( Utilities.DigestAlgorithm.SHA_256, body.password ).map(b=>(b+256).toString(16).slice(-2)).join(''); sheet.appendRow([id, body.name, body.dept, email, hash, body.phone||'', body.intro||'', 'member', new Date().toISOString()]); return out({success:true}); } // doGet 내 'login' 액션 case 'login': { const sheet = ss.getSheetByName('members'); const rows = sheet.getDataRange().getValues().slice(1); const hash = Utilities.computeDigest( Utilities.DigestAlgorithm.SHA_256, params.password ).map(b=>(b+256).toString(16).slice(-2)).join(''); const row = rows.find(r => r[3]===params.email && r[4]===hash); if (!row) return out({error:'이메일 또는 비밀번호가 올바르지 않습니다.'}); return out({success:true, user:{name:row[1],email:row[3],dept:row[2],role:row[7]}}); }
